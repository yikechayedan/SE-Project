class RegisterSerializer(serializers.ModelSerializer):
    password = serializers.CharField(write_only=True, required=True)
    code = serializers.CharField(write_only=True, required=True)  # ✅ 新增

    class Meta:
        model = User
        fields = ["id", "username", "password", "email", "phone", "code"]

修改点 2：在 serializer 里校验验证码
from .utils import check_code

def validate(self, attrs):
    email = attrs.get("email")
    code = attrs.get("code")

    if not check_code(email, code):
        raise serializers.ValidationError("邮箱验证码错误或已过期")

    return attrs


⭐ 这是关键点
校验逻辑一定要放在 serializer，而不是 view

修改点 3：create 时删掉 code
def create(self, validated_data):
    validated_data.pop("code")  # ✅ 不入库
    validated_data["password"] = make_password(validated_data["password"])
    return super().create(validated_data)

② views.py：新增「发送注册验证码」接口

📄 apps/users/views.py

新增一个 View（完全复用你们现有逻辑）
class RegisterSendCodeView(generics.GenericAPIView):
    permission_classes = [AllowAny]

    def post(self, request):
        email = request.data.get("email")
        if not email:
            return Response({"code": 400, "msg": "请提供邮箱"}, status=400)

        # 已注册直接拒绝
        if User.objects.filter(email=email).exists():
            return Response({"code": 400, "msg": "该邮箱已注册"}, status=400)

        code = generate_code()
        save_code(email, code)

        send_mail(
            subject="PolyMetric 注册验证码",
            message=f"您的注册验证码是：{code}，5分钟内有效。",
            from_email=None,
            recipient_list=[email],
            fail_silently=False,
        )

        return Response({"code": 200, "msg": "验证码已发送"})
